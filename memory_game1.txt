<script>
(function(){
  var EMO = ["🐶","🐱","🦊","🐼","🐸","🦁","🐵","🦄","🐙","🦖","🍕","🚀","🌈","⭐","⚽","🎸","🍦","🐝"];
  var pairs = 8, first = null, lock = false, moves = 0, found = 0, secs = 0, timer = null, started = false;
  var grid = document.getElementById("mm-grid");

  function fmt(s){ return Math.floor(s/60) + ":" + String(s%60).padStart(2,"0"); }
  function tick(){ secs++; document.getElementById("mm-time").textContent = fmt(secs); }
  function startTimer(){ if(!started){ started = true; timer = setInterval(tick, 1000); } }
  function stopTimer(){ clearInterval(timer); }

  function build(){
    stopTimer(); first = null; lock = false; moves = 0; found = 0; secs = 0; started = false;
    document.getElementById("mm-moves").textContent = "0";
    document.getElementById("mm-time").textContent = "0:00";
    document.getElementById("mm-pairs").textContent = "0/" + pairs;
    document.getElementById("mm-win").style.display = "none";
    grid.className = pairs <= 6 ? "c3" : (pairs <= 8 ? "c4" : "c5");
    var deck = EMO.slice(0, pairs).concat(EMO.slice(0, pairs));
    for (var i = deck.length - 1; i > 0; i--){ var j = Math.floor(Math.random()*(i+1)); var t = deck[i]; deck[i] = deck[j]; deck[j] = t; }
    grid.innerHTML = "";
    deck.forEach(function(e){
      var c = document.createElement("button");
      c.className = "mm-card";
      c.dataset.e = e;
      c.innerHTML = '<div class="mm-inner"><div class="mm-face mm-back">?</div><div class="mm-face mm-front" style="font-size: 26px;">'+e+'</div></div>';
      c.addEventListener("click", flip);
      grid.appendChild(c);
    });
  }

  function flip(){
    if (lock || this.classList.contains("up")) return;
    startTimer();
    this.classList.add("up");
    if (!first){ first = this; return; }
    moves++; document.getElementById("mm-moves").textContent = moves;
    if (first.dataset.e === this.dataset.e){
      first.classList.add("done"); this.classList.add("done");
      first = null; found++;
      document.getElementById("mm-pairs").textContent = found + "/" + pairs;
      if (found === pairs){
        stopTimer();
        document.getElementById("mm-winline").textContent = "Finished in " + moves + " moves and " + fmt(secs) + ".";
        document.getElementById("mm-win").style.display = "block";
      }
    } else {
      lock = true;
      var a = first, b = this; first = null;
      setTimeout(function(){ a.classList.remove("up"); b.classList.remove("up"); lock = false; }, 700);
    }
  }

  document.querySelectorAll("#mm-diff button").forEach(function(btn){
    btn.addEventListener("click", function(){
      document.querySelectorAll("#mm-diff button").forEach(function(x){ x.classList.remove("on"); });
      btn.classList.add("on");
      pairs = parseInt(btn.dataset.p, 10);
      build();
    });
  });
  document.getElementById("mm-restart").addEventListener("click", build);
  var th = document.getElementById("mm-theme");
  if (window.matchMedia && window.matchMedia("(prefers-color-scheme: dark)").matches) document.body.classList.add("dark");
  th.addEventListener("click", function(){
    document.body.classList.toggle("dark");
    th.textContent = document.body.classList.contains("dark") ? "Light mode" : "Dark mode";
  });
  if (document.body.classList.contains("dark")) th.textContent = "Light mode";
  build();
})();
</script>
